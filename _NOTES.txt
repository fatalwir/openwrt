/etc/init.d/rtc-osf-fix

#!/bin/sh /etc/rc.common
START=99
start() {
    I2CBUS=0        # change if needed
    ADDR=0x68

    STATUS=$(i2cget -y $I2CBUS $ADDR 0x0F)
    if [ $((STATUS & 0x80)) -ne 0 ]; then
        logger -t rtc-osf "RTC OSF flag set, resetting time"
        # Wait for NTP to set system time
        sleep 5
        hwclock --systohc
    fi
}


chmod +x /etc/init.d/rtc-osf-fix
/etc/init.d/rtc-osf-fix enable



----------------------- ADC Readout ------------------------------
while true; do
    echo ""

    raw1=$(cat /sys/bus/iio/devices/iio:device0/in_voltage0_raw)
    scale1=$(cat /sys/bus/iio/devices/iio:device0/in_voltage0_scale)
    voltage1=$(awk "BEGIN {printf \"%.2f\", $raw1 * $scale1 * 16}")
    echo "VBUS: $voltage1 V"

    raw2=$(cat /sys/bus/iio/devices/iio:device0/in_voltage1_raw)
    scale2=$(cat /sys/bus/iio/devices/iio:device0/in_voltage1_scale)
    voltage2=$(awk "BEGIN {printf \"%.2f\", $raw2 * $scale2 * 2.022}")
    echo "VBAT: $voltage2 V"
    sleep 1
done
